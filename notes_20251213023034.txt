  <script>
    // Simple confetti system
    (function(){
      const canvas = document.getElementById('confetti');
      const ctx = canvas.getContext('2d');
      let W = canvas.width = innerWidth;
      let H = canvas.height = innerHeight;
      window.addEventListener('resize', () => {
        W = canvas.width = innerWidth;
        H = canvas.height = innerHeight;
      });

      function rand(min,max){ return Math.random()*(max-min)+min; }

      class Particle {
        constructor(){
          this.reset();
        }
        reset(){
          this.x = rand(0,W);
          this.y = rand(-H,0);
          this.w = rand(6,14);
          this.h = rand(8,18);
          this.vx = rand(-1.5,1.5);
          this.vy = rand(2,6);
          this.spin = rand(-0.2,0.2);
          this.angle = rand(0,Math.PI*2);
          this.color = ["#ff6b80","#ffb3c6","#ffd6e0","#e33451","#ff8aa2"][Math.floor(rand(0,5))];
          this.opacity = rand(0.9,1);
        }
        update(){
          this.x += this.vx;
          this.y += this.vy;
          this.angle += this.spin;
          if(this.y > H + 50) this.reset();
        }
        draw(ctx){
          ctx.save();
          ctx.translate(this.x,this.y);
          ctx.rotate(this.angle);
          ctx.globalAlpha = this.opacity;
          ctx.fillStyle = this.color;
          ctx.fillRect(-this.w/2, -this.h/2, this.w, this.h);
          ctx.restore();
        }
      }

      let particles = [];
      let animId = null;
      function startConfetti(count=120){
        particles = [];
        for(let i=0;i<count;i++) particles.push(new Particle());
        let ticks = 0;
        function loop(){
          ticks++;
          ctx.clearRect(0,0,W,H);
          for(let p of particles){
            p.update();
            p.draw(ctx);
          }
          animId = requestAnimationFrame(loop);
        }
        if(animId) cancelAnimationFrame(animId);
        loop();
        // auto stop after 6s
        setTimeout(stopConfetti, 6000);
      }
      function stopConfetti(){
        if(animId) cancelAnimationFrame(animId);
        ctx.clearRect(0,0,W,H);
      }

      // Buttons and modal logic
      const yesBtn = document.getElementById('yesBtn');
      const noBtn = document.getElementById('noBtn');
      const modal = document.getElementById('modal');
      const closeBtn = document.getElementById('closeBtn');

      yesBtn.addEventListener('click', () => {
        // open modal, run confetti
        modal.classList.add('open');
        startConfetti(140);
        // change modal message optionally
        document.getElementById('modalMsg').textContent = "তুমি 'হ্যাঁ' বললে আমার কাছে সবচেয়ে বড় উপহার। আমি রাবসারি করে রাখলাম — চল একসাথে থাকি।";
      });

      closeBtn.addEventListener('click', () => {
        modal.classList.remove('open');
      });

      noBtn.addEventListener('click', () => {
        // playful nudge: move the "No" button randomly to make it cute
        for(let i=0;i<6;i++){
          setTimeout(()=>{
            noBtn.style.transform = `translate(${Math.random()*80-40}px, ${Math.random()*20-10}px)`;
          }, i*80);
        }
        setTimeout(()=>{ noBtn.style.transform = 'translate(0,0)'; }, 700);
      });

      // Accessibility: Enter key triggers "yes" if focused
      document.addEventListener('keydown', (e)=>{
        if(e.key === 'Enter') yesBtn.click();
      });

    })();
  </script>
